Random kütüphanesi: 

bismillah
ssfonksiyon mt(n){
    ssdeğişken a = ssmatematik.rastgele();
    ssdeğişken b = a * n;
    ssdeğişken c = ssmatematik.yuvarla(b);
    ssdöndür c;
}
ssdeğişken d = mt(5);
ssyazdır(d);

ssdeğişken kütüphaneadi2 = "Random İl Kütüphanesi";

ssfonksiyon randomil(){
ssdeğişken iller = ⌞"İstanbul"→"İzmir"→"Ankara"→"Kocaeli"→"Bursa"→"Ordu"→"Sivas"→"Manisa"→
"Trabzon"→"Edirne"→"Karabük"→"Erzincan"→"Burdur"→"Batman"⌝;
ssdeğişken random1 = ssmatematik.rastgele();
ssdeğişken illeruzunluk = sscetvel(iller);
ssdeğişken random3 = random1 * illeruzunluk;
ssdeğişken random2 = ssmatematik.yukari(random3);
ssdeğişken random = random2 - 1;
ssdeğişken secili = iller⌞random⌝;
ssdöndür secili;
}
